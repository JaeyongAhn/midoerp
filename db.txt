create table auth(
id int auto_increment primary key,
email varchar(255),
passwd varchar(255),
created_at timestamp default current_timestamp,
index(email)
);

create table account_id(
id int auto_increment primary key,
shorten varchar(100),
last_account timestamp,
created_at timestamp default current_timestamp,
index(shorten)
);

create table account(
id int auto_increment primary key,
account_id int,
trans_date timestamp,
chulguem long,
ipguem long,
janeak long,
naeyong varchar(100),
memo varchar(255),
tag1 varchar(100),
tag2 varchar(100),
tag3 varchar(100),
index(account_id),
index(trans_date),
index(tag1)
);

create table tags(
id int auto_increment primary key,
`mode` int,
name varchar(100) unique
);
